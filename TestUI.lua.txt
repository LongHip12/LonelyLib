local function getProperty(instance, property, default)
    local value = instance:GetAttribute(property)
    if value == nil then
        return default
    end
    return value
end

local function create(className, properties, parent)
    local instance = Instance.new(className)
    for property, value in pairs(properties or {}) do
        instance[property] = value
    end
    if parent then
        instance.Parent = parent
    end
    return instance
end

local function corner(parent, radius)
    return create("UICorner", {
        CornerRadius = UDim.new(0, radius or 10)
    }, parent)
end

local function stroke(parent, color, transparency, thickness)
    return create("UIStroke", {
        Color = color or Themes["Stroke Color"],
        Transparency = transparency or 0.5,
        Thickness = thickness or 1
    }, parent)
end

local function gradient(parent, first, second, rotation)
    local result = create("UIGradient", {
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, first or Themes["Button Color"]),
            ColorSequenceKeypoint.new(0.55, second or Themes["Button Color Bright"]),
            ColorSequenceKeypoint.new(1, Themes["Button Color Dark"])
        }),
        Rotation = rotation or -57
    }, parent)
    return result
end

local function text(parent, value, properties)
    local config = {
        BackgroundTransparency = 1,
        Text = value or "",
        TextColor3 = Themes["Text Color"],
        Font = Enum.Font.Gotham,
        TextSize = 13,
        RichText = true,
        TextWrapped = true,
        TextXAlignment = Enum.TextXAlignment.Left,
        TextYAlignment = Enum.TextYAlignment.Center
    }
    for property, setting in pairs(properties or {}) do
        config[property] = setting
    end
    return create("TextLabel", config, parent)
end

local function image(parent, asset, properties)
    local config = {
        BackgroundTransparency = 1,
        Image = asset or "",
        ScaleType = Enum.ScaleType.Fit
    }
    for property, setting in pairs(properties or {}) do
        config[property] = setting
    end
    return create("ImageLabel", config, parent)
end

local function setVisible(instance, visible)
    if instance and instance.Parent then
        instance.Visible = visible
    end
end

local function makeDraggable(handle, target)
    local dragging = false
    local dragStart
    local startPosition

    local function update(input)
        local delta = input.Position - dragStart
        target.Position = UDim2.new(
            startPosition.X.Scale,
            startPosition.X.Offset + delta.X,
            startPosition.Y.Scale,
            startPosition.Y.Offset + delta.Y
        )
    end

    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPosition = target.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            update(input)
        end
    end)
end

local function createPanel(parent, size, position, layoutOrder)
    local panel = create("Frame", {
        BackgroundColor3 = Themes["Panel Color"],
        BackgroundTransparency = Themes["Panel Transparency"],
        BorderSizePixel = 0,
        Size = size,
        Position = position or UDim2.new(),
        LayoutOrder = layoutOrder or 0
    }, parent)
    corner(panel, 10)
    local panelGradient = create("Frame", {
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        Size = UDim2.fromScale(1, 1)
    }, panel)
    corner(panelGradient, 10)
    gradient(panelGradient, Themes["Button Color Dark"], Themes["Panel Color"], -90)
    panelGradient.BackgroundTransparency = 0.92
    stroke(panel, Themes["Stroke Color"], 0.72, 1)
    return panel
end

end

local function normalizeConfig(config)
    if type(config) == "string" then
        return {Title = config}
    end
    return config or {}
end

local function setSide(instance, side)
    if instance and instance.SetAttribute then
        instance:SetAttribute("Side", side == "Right" and "Right" or "Left")
    end
    return instance
end

                    if options.Multi then

                        updateText()
                        if options.Callback then
                            options.Callback(currentValues)
                        end
                    else
                        selected = value
                        updateText()

            end

                popup.Visible = not popup.Visible

            end)
            setSide(holder, options.Side or section.Side)
            table.insert(section.Controls, holder)
            return {
                Frame = holder,

        function section:AddSlider(options)
            options = options or {}
            local holder = create("TextButton", {
                AutoButtonColor = false,
                BackgroundTransparency = 1,
                BorderSizePixel = 0,
                LayoutOrder = #section.Controls + 1,
                Size = UDim2.new(1, 0, 0, 58),
                Text = ""
            }, body)
            local minimum = tonumber(options.Min) or 0
            local maximum = tonumber(options.Max) or 100
            local increase = tonumber(options.Increase) or 1
            local amount = tonumber(options.Default) or minimum
            local titleLabel = text(holder, options.Title or "Slider", {
                Position = UDim2.fromScale(0.04, 0.02),
                Size = UDim2.fromScale(0.62, 0.32),
                TextSize = 11,
                Font = Enum.Font.GothamBold
            })
            local amountLabel = text(holder, tostring(amount), {
                AnchorPoint = Vector2.new(1, 0),
                Position = UDim2.fromScale(0.96, 0.02),
                Size = UDim2.fromScale(0.25, 0.32),
                TextSize = 11,
                TextXAlignment = Enum.TextXAlignment.Right
            })
            local sliderFrame = create("Frame", {
                BackgroundColor3 = Themes["Text Color"],
                BorderSizePixel = 0,
                Position = UDim2.fromScale(0.06, 0.61),
                Size = UDim2.fromScale(0.88, 0.1)
            }, holder)
            corner(sliderFrame, 6)
            local progress = create("Frame", {
                BackgroundColor3 = Themes["Slider Color"],
                BorderSizePixel = 0,
                Size = UDim2.fromScale(0, 1)
            }, sliderFrame)
            corner(progress, 6)
            local marker = image(holder, options.MarkerImage or Themes.Images["Slider Marker"], {
                AnchorPoint = Vector2.new(0.5, 0.5),
                Position = UDim2.fromScale(0.06, 0.66),
                Size = UDim2.fromScale(0.14, 0.42),
                ZIndex = 2
            })
            local function roundValue(value)
                local rounded = math.floor(value / increase + 0.5) * increase
                if increase < 1 then
                    local decimals = math.max(0, string.len(tostring(increase):match("%.(%d+)") or "1"))
                    rounded = tonumber(string.format("%." .. decimals .. "f", rounded))
                end
                return math.clamp(rounded, minimum, maximum)
            end
            local function setSlider(value, call)
                amount = roundValue(value)
                local alpha = maximum == minimum and 0 or (amount - minimum) / (maximum - minimum)
                progress.Size = UDim2.fromScale(alpha, 1)
                marker.Position = UDim2.fromScale(0.06 + alpha * 0.88, 0.66)
                amountLabel.Text = tostring(amount)
                if call and options.Callback then
                    options.Callback(amount)
                end
            end
            local function inputToValue(input)
                local alpha = math.clamp((input.Position.X - sliderFrame.AbsolutePosition.X) / sliderFrame.AbsoluteSize.X, 0, 1)
                return minimum + (maximum - minimum) * alpha
            end
            local sliderDragging = false
            sliderFrame.InputBegan:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                    sliderDragging = true
                    setSlider(inputToValue(input), true)
                end
            end)
            UserInputService.InputChanged:Connect(function(input)
                if sliderDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                    setSlider(inputToValue(input), true)
                end
            end)
            UserInputService.InputEnded:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                    sliderDragging = false
                end
            end)
            setSlider(amount, false)
            setSide(holder, options.Side or section.Side)
            table.insert(section.Controls, holder)
            return {
                Frame = holder,
                SetValue = function(_, value)
                    setSlider(value, true)
                end,
                GetValue = function()
                    return amount
                end
            }
        end

    function window:Toggle()
        main.Visible = not main.Visible
    end

    function window:Destroy()
        screenGui:Destroy()
    end

    function window:SetVisible(value)
        main.Visible = value == true
    end

    function window:GetVisible()
        return main.Visible
    end

    task.defer(function()
        if not mainTabsConfigured and #window.Tabs >= 3 then
            print("LonelyLib: SetMainTab was not called")
        end
    end)

    Library.Window = window
    return window
end

function Library:Notify(config)
    if Library.Window then
        return Library.Window:Notify(config)
    end
end

return Library
