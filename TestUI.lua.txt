local function protectGui(gui)
    local protected = false
    if type(gethui) == "function" then
        local holder = gethui()
        if holder then
            gui.Parent = holder
            protected = true
        end
    end
    if not protected then
        local coreGui = game:GetService("CoreGui")
        local success = pcall(function()
            gui.Parent = coreGui
        end)
        protected = success
    end
    if not protected then
        gui.Parent = Players.LocalPlayer:WaitForChild("PlayerGui")
    end
end

local function create(className, properties, parent)
    local object = Instance.new(className)
    for key, value in pairs(properties or {}) do
        pcall(function()
            object[key] = value
        end)
    end
    if parent then
        object.Parent = parent
    end
    return object
end

local function corner(parent, radius)
    return create("UICorner", {
        CornerRadius = UDim.new(0, radius or 8)
    }, parent)
end

local function stroke(parent, color, thickness, transparency)
    return create("UIStroke", {
        Color = color or Themes["Stroke Color"],
        Thickness = thickness or 1,
        Transparency = transparency or 0
    }, parent)
end

local function padding(parent, left, right, top, bottom)
    return create("UIPadding", {
        PaddingLeft = UDim.new(0, left or 0),
        PaddingRight = UDim.new(0, right or 0),
        PaddingTop = UDim.new(0, top or 0),
        PaddingBottom = UDim.new(0, bottom or 0)
    }, parent)
end

local function layout(parent, direction, paddingValue, horizontalAlignment)
    return create("UIListLayout", {
        FillDirection = direction or Enum.FillDirection.Vertical,
        SortOrder = Enum.SortOrder.LayoutOrder,
        Padding = UDim.new(0, paddingValue or 0),
        HorizontalAlignment = horizontalAlignment or Enum.HorizontalAlignment.Left
    }, parent)
end

local function tween(object, properties, duration)
    local info = TweenInfo.new(duration or 0.18, Enum.EasingStyle.Quint, Enum.EasingDirection.Out)
    TweenService:Create(object, info, properties):Play()
end

local function addImage(parent, image, size, position)
    local object = create("ImageLabel", {
        BackgroundTransparency = 1,
        Image = image or "",
        ImageColor3 = Themes["Image Color"],
        Size = size or UDim2.fromOffset(18, 18),
        Position = position or UDim2.fromOffset(0, 0),
        ScaleType = Enum.ScaleType.Fit,
        Visible = image ~= nil and image ~= ""
    }, parent)
    return object
end

local function setText(object, text)
    object.Text = text or ""
    object.RichText = true
end

local function normalizeSide(side)
    if side == "Right" or side == "right" then
        return "Right"
    end
    return "Left"
end

local function roundValue(value, increase)
    if type(increase) ~= "number" or increase <= 0 then
        return value
    end
    return math.floor((value / increase) + 0.5) * increase
end

local function valueText(value)
    if type(value) == "number" then
        if math.floor(value) == value then
            return tostring(math.floor(value))
        end
        return string.format("%.3f", value):gsub("0+$", ""):gsub("%.$", "")
    end
    return tostring(value or "")
end

addHover = function(button, normal, hover)
    button.MouseEnter:Connect(function()
        tween(button, {
            BackgroundColor3 = hover
        }, 0.14)
    end)
    button.MouseLeave:Connect(function()
        tween(button, {
            BackgroundColor3 = normal
        }, 0.14)
    end)
end

addDrag = function(handle, target)
    local dragging = false
    local dragStart
    local startPosition
    local dragInput
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPosition = target.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    handle.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            target.Position = UDim2.new(
                startPosition.X.Scale,
                startPosition.X.Offset + delta.X,
                startPosition.Y.Scale,
                startPosition.Y.Offset + delta.Y
            )
        end
    end)
end

function Window:_showPage(page)
    for _, tab in ipairs(self.Tabs) do
        tab.Page.Visible = tab == page
        tab.Button.BackgroundColor3 = tab == page and Themes["Button Color"] or Themes["Surface Color"]
        tab.Button.TextColor3 = tab == page and Themes["Text Color"] or Themes["Muted Text Color"]
    end
    self.HomeTab.Visible = page == nil
    self.CreditsTab.Visible = page == self.CreditsTab
    if page then
        self.PagesTitle.Text = page.Title
        self.PagesDescription.Text = page.Description
    end
end

function Window:_showHome()

function Window:SetMainTab(a, b, c)
    local tabs = { a, b, c }
    if #tabs ~= 3 or not a or not b or not c or a == b or a == c or b == c then
        print("[HoHo Library] SetMainTab requires exactly three different tabs.")
        return false
    end
    for _, tab in ipairs(tabs) do
        if tab.Window ~= self then
            print("[HoHo Library] SetMainTab received a tab from another window.")
            return false
        end
    end
    self.MainTabs = tabs
    self.MainTabsSet = true
    self:_buildHomeCards()
    return true
end

function Section:AddDropdown(config)
    config = config or {}
    local values = config.Values or {}
    local multi = config.Multi == true
    local selected = {}
    if multi then
        if type(config.Default) == "table" then
            for _, value in ipairs(config.Default) do
                selected[value] = true
            end
        elseif config.Default ~= nil then
            selected[config.Default] = true
        end
    else
        selected[1] = config.Default or values[1] or ""
    end
    local row = create("Frame", {
        Name = "Selector_Single",
        BackgroundColor3 = Themes["Background Color"],
        BorderSizePixel = 0,
        Size = UDim2.new(1, 0, 0, 58),
        LayoutOrder = #self.Elements + 1
    }, self.Content)
    corner(row, 7)
    stroke(row, Themes["Muted Stroke Color"], 1, 0.1)
    local title = create("TextLabel", {
        BackgroundTransparency = 1,
        Font = Enum.Font.GothamMedium,
        TextColor3 = Themes["Text Color"],
        TextSize = 11,
        TextXAlignment = Enum.TextXAlignment.Left,
        Position = UDim2.fromOffset(10, 7),
        Size = UDim2.new(1, -20, 0, 17)
    }, row)
    setText(title, config.Title or "Dropdown")
    local display = create("TextButton", {
        AutoButtonColor = false,
        BackgroundColor3 = Themes["Surface Color"],
        BorderSizePixel = 0,
        Font = Enum.Font.Gotham,
        TextColor3 = Themes["Muted Text Color"],
        TextSize = 10,
        TextXAlignment = Enum.TextXAlignment.Left,
        Position = UDim2.fromOffset(8, 29),
        Size = UDim2.new(1, -16, 0, 23),
        Text = ""
    }, row)
    corner(display, 5)
    padding(display, 9, 9, 0, 0)
    local arrow = create("TextLabel", {
        BackgroundTransparency = 1,
        Font = Enum.Font.GothamBold,
        TextColor3 = Themes["Button Color"],
        TextSize = 12,
        Text = "›",
        TextXAlignment = Enum.TextXAlignment.Center,
        Position = UDim2.new(1, -25, 0, 0),
        Size = UDim2.fromOffset(20, 23)
    }, display)
    local popup
    local api = {}
    local function getDisplay()
        if multi then
            local list = {}
            for _, value in ipairs(values) do
                if selected[value] then
                    table.insert(list, tostring(value))
                end
            end
            return #list > 0 and table.concat(list, ", ") or "Select..."
        end
        return valueText(selected[1])
    end
    local function closePopup()
        if popup then
            popup:Destroy()
            popup = nil
        end
    end
    local function choose(value)
        if multi then
            selected[value] = not selected[value]
        else
            selected[1] = value
            closePopup()
        end
        display.Text = getDisplay()
        if type(config.Callback) == "function" then
            if multi then
                local result = {}
                for _, option in ipairs(values) do
                    if selected[option] then
                        table.insert(result, option)
                    end
                end
                config.Callback(result)
            else
                config.Callback(selected[1])
            end
        end
    end
    local function openPopup()
        closePopup()
        popup = create("Frame", {
            Name = "List",
            BackgroundColor3 = Themes["Surface Color"],
            BorderSizePixel = 0,
            Position = UDim2.new(0, 0, 1, 4),
            Size = UDim2.new(1, 0, 0, math.min(170, math.max(34, #values * 28 + 8))),
            ZIndex = 30
        }, row)
        corner(popup, 6)
        stroke(popup, Themes["Button Color"], 1, 0.1)
        padding(popup, 5, 5, 5, 5)
        layout(popup, Enum.FillDirection.Vertical, 4)
        for index, value in ipairs(values) do
            local option = create("TextButton", {
                AutoButtonColor = false,
                BackgroundColor3 = selected[value] and Themes["Button Color"] or Themes["Surface Color 2"],
                BorderSizePixel = 0,
                Font = Enum.Font.Gotham,
                TextColor3 = Themes["Text Color"],
                TextSize = 10,
                TextXAlignment = Enum.TextXAlignment.Left,
                Text = "  " .. tostring(value),
                Size = UDim2.new(1, 0, 0, 24),
                LayoutOrder = index,
                ZIndex = 31
            }, popup)
            corner(option, 4)
            option.MouseButton1Click:Connect(function()
                choose(value)
                if multi then
                    option.BackgroundColor3 = selected[value] and Themes["Button Color"] or Themes["Surface Color 2"]
                end
            end)
            addHover(option, selected[value] and Themes["Button Color"] or Themes["Surface Color 2"], Themes["Button Color"])
        end
    end
    display.MouseButton1Click:Connect(function()
        if popup then
            closePopup()
        else
            openPopup()
        end
    end)
    display.Text = getDisplay()
    api.Instance = row
    api.Value = function()
        if multi then
            local result = {}
            for _, option in ipairs(values) do
                if selected[option] then
                    table.insert(result, option)
                end
            end
            return result
        end
        return selected[1]
    end
    api.Set = function(_, value)
        if multi then
            selected = {}
            for _, option in ipairs(value or {}) do
                selected[option] = true
            end
        else
            selected[1] = value
        end
        display.Text = getDisplay()
    end
    table.insert(self.Elements, api)
    return api
end

function Section:AddSlider(config)
    config = config or {}
    local minimum = tonumber(config.Min) or 0
    local maximum = tonumber(config.Max) or 100
    local increase = tonumber(config.Increase) or 1
    local value = math.clamp(tonumber(config.Default) or minimum, minimum, maximum)
    value = roundValue(value, increase)
    local row = create("Frame", {
        Name = "Slider",
        BackgroundColor3 = Themes["Background Color"],
        BorderSizePixel = 0,
        Size = UDim2.new(1, 0, 0, 60),
        LayoutOrder = #self.Elements + 1
    }, self.Content)
    corner(row, 7)
    stroke(row, Themes["Muted Stroke Color"], 1, 0.1)
    local title = create("TextLabel", {
        BackgroundTransparency = 1,
        Font = Enum.Font.GothamMedium,
        TextColor3 = Themes["Text Color"],
        TextSize = 11,
        TextXAlignment = Enum.TextXAlignment.Left,
        Position = UDim2.fromOffset(10, 7),
        Size = UDim2.new(1, -75, 0, 17)
    }, row)
    setText(title, config.Title or "Slider")
    local current = create("TextLabel", {
        BackgroundTransparency = 1,
        Font = Enum.Font.GothamBold,
        TextColor3 = Themes["Button Color"],
        TextSize = 11,
        TextXAlignment = Enum.TextXAlignment.Right,
        Position = UDim2.new(1, -65, 0, 7),
        Size = UDim2.fromOffset(55, 17)
    }, row)
    local bar = create("Frame", {
        BackgroundColor3 = Themes["Toggle Off Color"],
        BorderSizePixel = 0,
        Position = UDim2.fromOffset(10, 38),
        Size = UDim2.new(1, -20, 0, 7)
    }, row)
    corner(bar, 4)
    local fill = create("Frame", {
        BackgroundColor3 = Themes["Slider Color"],
        BorderSizePixel = 0,
        Size = UDim2.new(0, 0, 1, 0)
    }, bar)
    corner(fill, 4)
    local knob = create("Frame", {
        BackgroundColor3 = Themes["Text Color"],
        BorderSizePixel = 0,
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0, 0, 0.5, 0),
        Size = UDim2.fromOffset(12, 12)
    }, bar)
    corner(knob, 6)
    local api = {}
    local function setValue(newValue, call)
        value = math.clamp(roundValue(tonumber(newValue) or minimum, increase), minimum, maximum)
        local percent = maximum == minimum and 0 or (value - minimum) / (maximum - minimum)
        current.Text = valueText(value)
        fill.Size = UDim2.new(percent, 0, 1, 0)
        knob.Position = UDim2.new(percent, 0, 0.5, 0)
        if type(config.Callback) == "function" and call then
            config.Callback(value)
        end
    end
    local dragging = false
    local function updateFromInput(input)
        local percent = math.clamp((input.Position.X - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
        setValue(minimum + (maximum - minimum) * percent, true)
    end
    bar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            updateFromInput(input)
        end
    end)
    bar.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            updateFromInput(input)
        end
    end)
    setValue(value, false)
    api.Instance = row
    api.Value = function()
        return value
    end
    api.Set = function(_, newValue)
        setValue(newValue, true)
    end
    table.insert(self.Elements, api)
    return api
end
